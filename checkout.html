import { serve } from "https://deno.land/std@0.168.0/http/server.ts" import { createClient } from 'https://esm.sh/@supabase/supabase-js@2' const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type', } serve(async (req) => { if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders }) try { const supabaseClient = createClient( Deno.env.get('SUPABASE_URL') ?? '', Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? '' ) const body = await req.json() const { pedido_id, metodo, parcelas, cartao_token, cartao_bandeira } = body console.log('📦 Recebido:', { pedido_id, metodo, parcelas, bandeira: cartao_bandeira }) if (!pedido_id) throw new Error('pedido_id obrigatório') const { data: pedido } = await supabaseClient.from('pedidos').select('*').eq('id', pedido_id).single() if (!pedido) throw new Error('Pedido não encontrado') const { data: gateway } = await supabaseClient.from('gateways').select('*').eq('ativo', true).limit(1).single() if (!gateway) throw new Error('Gateway não ativo') const accessToken = gateway.credenciais?.access_token if (!accessToken) throw new Error('Access Token ausente') let resultado = null if (metodo === 'credit_card') { if (!cartao_token) throw new Error('Token do cartão ausente') // ⭐ PASSO CRÍTICO: descobrir o payment_method_id REAL let paymentMethodId = cartao_bandeira || null // Se a bandeira não veio do frontend, consulta no token if (!paymentMethodId) { console.log('🔎 Bandeira não recebida. Consultando token no MP...') const tokenResp = await fetch(`https://api.mercadopago.com/v1/card_tokens/${cartao_token}`, { headers: { 'Authorization': `Bearer ${accessToken}` } }) if (tokenResp.ok) { const tokenData = await tokenResp.json() paymentMethodId = tokenData.payment_method_id console.log('🔎 Bandeira obtida:', paymentMethodId) } else { throw new Error('Token inválido ou expirado. Tente novamente.') } } if (!paymentMethodId) throw new Error('Não foi possível identificar a bandeira do cartão') const payload = { transaction_amount: Number(pedido.total), description: `Pedido ${pedido_id} - S7TE`, token: cartao_token, installments: Number(parcelas) || 1, payment_method_id: paymentMethodId, // ⭐ USA A BANDEIRA REAL (visa, master, elo) payer: { email: String(pedido.cliente_email), first_name: String(pedido.cliente_nome).split(' ')[0] || 'Cliente', last_name: String(pedido.cliente_nome).split(' ').slice(1).join(' ') || '', identification: { type: pedido.cliente_cpf.length === 14 ? 'CNPJ' : 'CPF', number: String(pedido.cliente_cpf).replace(/\D/g, '') } } } console.log('💳 Enviando para MP:', { payment_method_id: paymentMethodId, valor: payload.transaction_amount }) const response = await fetch('https://api.mercadopago.com/v1/payments', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${accessToken}`, 'X-Idempotency-Key': `pedido-${pedido_id}-${Date.now()}` }, body: JSON.stringify(payload) }) const mpData = await response.json() console.log('📥 Resposta MP:', response.status, mpData.status) if (!response.ok) { throw new Error(`MP rejeitou (HTTP ${response.status}): ${mpData.message || 'erro desconhecido'}`) } resultado = mpData if (mpData.id) { const statusMap: Record = { 'approved': 'pago', 'authorized': 'pago', 'pending': 'aguardando_pagamento', 'in_process': 'aguardando_pagamento', 'rejected': 'recusado', 'cancelled': 'cancelado' } await supabaseClient.from('pedidos').update({ gateway_payment_id: String(mpData.id), status: statusMap[mpData.status] || 'aguardando_pagamento', cartao_bandeira: mpData.payment_method_id || paymentMethodId, cartao_ultimo4: mpData.card?.last_four_digits || null }).eq('id', pedido_id) if (mpData.status === 'rejected') { throw new Error(`Pagamento recusado: ${mpData.status_detail || 'verifique o cartão'}`) } } } else if (metodo === 'pix') { const response = await fetch('https://api.mercadopago.com/v1/payments', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${accessToken}`, 'X-Idempotency-Key': `pix-${pedido_id}-${Date.now()}` }, body: JSON.stringify({ transaction_amount: Number(pedido.total), description: `Pedido ${pedido_id} - S7TE`, payment_method_id: 'pix', payer: { email: String(pedido.cliente_email), first_name: String(pedido.cliente_nome).split(' ')[0] || 'Cliente', identification: { type: pedido.cliente_cpf.length === 14 ? 'CNPJ' : 'CPF', number: String(pedido.cliente_cpf).replace(/\D/g, '') } } }) }) const mpData = await response.json() if (!response.ok) throw new Error(`MP rejeitou PIX: ${mpData.message || 'erro'}`) resultado = mpData if (mpData.id) { await supabaseClient.from('pedidos').update({ gateway_payment_id: String(mpData.id), status: 'aguardando_pagamento', pix_qr_code: mpData.point_of_interaction?.transaction_data?.qr_code || null, pix_qr_code_base64: mpData.point_of_interaction?.transaction_data?.qr_code_base64 || null, pix_copia_e_cola: mpData.point_of_interaction?.transaction_data?.qr_code || null }).eq('id', pedido_id) } } else { throw new Error(`Método não suportado: ${metodo}`) } return new Response(JSON.stringify({ success: true, status: resultado?.status, payment_id: resultado?.id, qr_code_base64: resultado?.point_of_interaction?.transaction_data?.qr_code_base64, copia_e_cola: resultado?.point_of_interaction?.transaction_data?.qr_code }), { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }) } catch (error) { console.error('💥 ERRO:', error.message) return new Response( JSON.stringify({ success: false, error: error.message }), { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } } ) } })
